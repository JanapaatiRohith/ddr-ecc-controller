========== 1. ECC ENCODER/DECODER ==========
`timescale 1ns/1ps

// 64-bit SECDED encoder: 64 data + 7 Hamming parity + 1 overall parity
module secded_64_72_encoder (
    input  wire [63:0] data_in,
    output reg  [71:0] codeword_out
);
    integer pos;
    integer data_idx;
    integer p;
    reg parity_bit;

    always @* begin
        codeword_out = 72'b0;
        data_idx = 0;

        // Place data in non-power-of-two positions 1 through 71.
        for (pos = 1; pos <= 71; pos = pos + 1) begin
            if ((pos != 1) && (pos != 2) && (pos != 4) &&
                (pos != 8) && (pos != 16) && (pos != 32) &&
                (pos != 64)) begin
                codeword_out[pos-1] = data_in[data_idx];
                data_idx = data_idx + 1;
            end
        end

        // Generate seven Hamming parity bits.
        for (p = 0; p < 7; p = p + 1) begin
            parity_bit = 1'b0;
            for (pos = 1; pos <= 71; pos = pos + 1) begin
                if ((pos & (1 << p)) != 0)
                    parity_bit = parity_bit ^ codeword_out[pos-1];
            end
            codeword_out[(1 << p)-1] = parity_bit;
        end

        // Overall even parity.
        codeword_out[71] = ^codeword_out[70:0];
    end
endmodule


module secded_64_72_decoder (
    input  wire [71:0] codeword_in,
    output reg  [63:0] data_out,
    output reg         single_error,
    output reg         double_error,
    output reg  [6:0]  syndrome
);
    integer pos;
    integer data_idx;
    integer p;
    reg parity_bit;
    reg overall_parity;
    reg [71:0] corrected_codeword;

    always @* begin
        syndrome = 7'b0;
        data_out = 64'b0;
        single_error = 1'b0;
        double_error = 1'b0;
        corrected_codeword = codeword_in;
        overall_parity = ^codeword_in;

        // Calculate syndrome.
        for (p = 0; p < 7; p = p + 1) begin
            parity_bit = 1'b0;
            for (pos = 1; pos <= 71; pos = pos + 1) begin
                if ((pos & (1 << p)) != 0)
                    parity_bit = parity_bit ^ codeword_in[pos-1];
            end
            syndrome[p] = parity_bit;
        end

        // Classify and correct errors.
        if ((syndrome != 7'd0) && overall_parity) begin
            single_error = 1'b1;
            if (syndrome <= 7'd71)
                corrected_codeword[syndrome-1'b1] =
                    ~corrected_codeword[syndrome-1'b1];
        end
        else if ((syndrome == 7'd0) && overall_parity) begin
            // Overall parity-bit error.
            single_error = 1'b1;
            corrected_codeword[71] = ~corrected_codeword[71];
        end
        else if ((syndrome != 7'd0) && !overall_parity) begin
            double_error = 1'b1;
        end

        // Extract the original 64-bit payload.
        data_idx = 0;
        for (pos = 1; pos <= 71; pos = pos + 1) begin
            if ((pos != 1) && (pos != 2) && (pos != 4) &&
                (pos != 8) && (pos != 16) && (pos != 32) &&
                (pos != 64)) begin
                data_out[data_idx] = corrected_codeword[pos-1];
                data_idx = data_idx + 1;
            end
        end
    end
endmodule

========== 3. DDR CONTROLLER RTL ==========
`timescale 1ns/1ps
// Educational DDR3-like controller for RTL simulation and open-source RTL-to-GDS flow.
// IMPORTANT: This is a simplified controller model, not a replacement for a DDR PHY/MIG.
//
// Address map (ADDR_WIDTH=12):
//   addr[11:10] : bank[1:0]
//   addr[ 9: 4] : row[5:0]
//   addr[ 3: 0] : column[3:0]
//
// Policy: single outstanding transaction, closed-page operation.
// Timing is expressed in controller clock cycles for simulation/synthesis.

module ddr_controller #(
    parameter integer ADDR_WIDTH       = 12,
    parameter integer ROW_WIDTH        = 6,
    parameter integer COL_WIDTH        = 4,
    parameter integer DATA_WIDTH       = 16,
    parameter integer BANK_WIDTH       = 2,
    parameter integer T_RCD            = 2,
    parameter integer T_RP             = 2,
    parameter integer T_RFC            = 6,
    parameter integer T_WR             = 2,
    parameter integer READ_LATENCY     = 3,
    parameter integer INIT_CYCLES      = 8,
    parameter integer REFRESH_INTERVAL = 40
)(
    input  wire                         clk,
    input  wire                         rst_n,

    // Simple user command interface
    input  wire                         cmd_valid,
    output wire                         cmd_ready,
    input  wire                         cmd_write,
    input  wire [ADDR_WIDTH-1:0]        cmd_addr,
    input  wire [DATA_WIDTH-1:0]        cmd_wdata,

    output reg                          rsp_valid,
    output reg  [DATA_WIDTH-1:0]        rsp_rdata,

    // Simplified DDR command/address interface
    output reg                          ddr_ck,
    output reg                          ddr_cke,
    output reg                          ddr_cs_n,
    output reg                          ddr_ras_n,
    output reg                          ddr_cas_n,
    output reg                          ddr_we_n,
    output reg  [BANK_WIDTH-1:0]        ddr_ba,
    output reg  [ROW_WIDTH-1:0]         ddr_a,

    // Simplified bidirectional DQ split into input/output/oe for ASIC flow
    output reg  [DATA_WIDTH-1:0]         ddr_dq_o,
    output reg                          ddr_dq_oe,
    input  wire [DATA_WIDTH-1:0]         ddr_dq_i,

    output reg                          init_calib_complete,
    output reg                          busy,
    output reg                          refresh_event
);

    localparam [4:0]
        S_RESET     = 5'd0,
        S_INIT      = 5'd1,
        S_IDLE      = 5'd2,
        S_PRE       = 5'd3,
        S_WAIT_RP   = 5'd4,
        S_ACT       = 5'd5,
        S_WAIT_RCD  = 5'd6,
        S_READ      = 5'd7,
        S_READ_WAIT = 5'd8,
        S_WRITE     = 5'd9,
        S_WAIT_WR   = 5'd10,
        S_PRE_DONE  = 5'd11,
        S_REFRESH   = 5'd12,
        S_WAIT_RFC  = 5'd13,
        S_RESP      = 5'd14;

    reg [4:0] state;

    reg [15:0] timer;
    reg [15:0] refresh_timer;
    reg         refresh_pending;

    reg [BANK_WIDTH-1:0] target_bank;
    reg [ROW_WIDTH-1:0]  target_row;
    reg [COL_WIDTH-1:0]  target_col;
    reg [DATA_WIDTH-1:0] target_wdata;
    reg                  target_write;

    reg [DATA_WIDTH-1:0] read_data_hold;

    reg [3:0] open_valid;
    reg [ROW_WIDTH-1:0] open_row [0:3];

    wire [BANK_WIDTH-1:0] addr_bank = cmd_addr[ADDR_WIDTH-1 -: BANK_WIDTH];
    wire [ROW_WIDTH-1:0]  addr_row  = cmd_addr[COL_WIDTH +: ROW_WIDTH];
    wire [COL_WIDTH-1:0]  addr_col  = cmd_addr[COL_WIDTH-1:0];

    wire target_page_hit =
        open_valid[target_bank] && (open_row[target_bank] == target_row);

    // A command must never be advertised as ready in the same cycle that
    // automatic refresh is due, because refresh has priority in S_IDLE.
    wire refresh_due = (refresh_timer >= REFRESH_INTERVAL-1);

    assign cmd_ready = (state == S_IDLE) &&
                       init_calib_complete &&
                       !busy &&
                       !refresh_due;

    // Command encoding:
    // ACT  : RAS=0 CAS=1 WE=1
    // READ : RAS=1 CAS=0 WE=1
    // WRITE: RAS=1 CAS=0 WE=0
    // PRE  : RAS=0 CAS=1 WE=0
    // REF  : RAS=0 CAS=0 WE=1
    task automatic nop_cmd;
    begin
        ddr_cs_n   = 1'b0;
        ddr_ras_n  = 1'b1;
        ddr_cas_n  = 1'b1;
        ddr_we_n   = 1'b1;
        ddr_ba     = {BANK_WIDTH{1'b0}};
        ddr_a      = {ROW_WIDTH{1'b0}};
    end
    endtask

    always @(*) begin
        // Safe defaults
        ddr_ck     = clk;
        ddr_cke    = init_calib_complete;
        ddr_cs_n   = 1'b0;
        ddr_ras_n  = 1'b1;
        ddr_cas_n  = 1'b1;
        ddr_we_n   = 1'b1;
        ddr_ba     = target_bank;
        ddr_a      = {ROW_WIDTH{1'b0}};
        ddr_dq_o   = target_wdata;
        ddr_dq_oe  = 1'b0;

        case (state)
            S_ACT: begin
                ddr_ras_n = 1'b0;
                ddr_ba    = target_bank;
                ddr_a     = target_row;
            end

            S_READ: begin
                ddr_cas_n = 1'b0;
                ddr_ba    = target_bank;
                ddr_a     = {{(ROW_WIDTH-COL_WIDTH){1'b0}}, target_col};
            end

            S_WRITE: begin
                ddr_cas_n = 1'b0;
                ddr_we_n  = 1'b0;
                ddr_ba    = target_bank;
                ddr_a     = {{(ROW_WIDTH-COL_WIDTH){1'b0}}, target_col};
                ddr_dq_o  = target_wdata;
                ddr_dq_oe = 1'b1;
            end

            S_PRE, S_PRE_DONE: begin
                ddr_ras_n = 1'b0;
                ddr_we_n  = 1'b0;
                ddr_ba    = target_bank;
                ddr_a     = {ROW_WIDTH{1'b0}};
                ddr_a[10%ROW_WIDTH] = 1'b0;
            end

            S_REFRESH: begin
                ddr_ras_n = 1'b0;
                ddr_cas_n = 1'b0;
                ddr_ba    = {BANK_WIDTH{1'b0}};
                ddr_a     = {ROW_WIDTH{1'b0}};
            end

            default: begin
            end
        endcase
    end

    integer i;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state               <= S_RESET;
            timer               <= 16'd0;
            refresh_timer        <= 16'd0;
            refresh_pending     <= 1'b0;
            target_bank         <= {BANK_WIDTH{1'b0}};
            target_row          <= {ROW_WIDTH{1'b0}};
            target_col          <= {COL_WIDTH{1'b0}};
            target_wdata        <= {DATA_WIDTH{1'b0}};
            target_write        <= 1'b0;
            read_data_hold      <= {DATA_WIDTH{1'b0}};
            rsp_valid           <= 1'b0;
            rsp_rdata           <= {DATA_WIDTH{1'b0}};
            init_calib_complete <= 1'b0;
            busy                <= 1'b0;
            refresh_event       <= 1'b0;
            for (i = 0; i < 4; i = i + 1) begin
                open_valid[i] <= 1'b0;
                open_row[i]   <= {ROW_WIDTH{1'b0}};
            end
        end else begin
            rsp_valid     <= 1'b0;
            refresh_event <= 1'b0;

            // Refresh interval counter pauses while maintenance is running.
            if (init_calib_complete &&
                state != S_REFRESH && state != S_WAIT_RFC) begin
                if (refresh_timer < REFRESH_INTERVAL)
                    refresh_timer <= refresh_timer + 16'd1;
            end

            case (state)

                S_RESET: begin
                    busy <= 1'b1;
                    timer <= 16'd0;
                    state <= S_INIT;
                end

                S_INIT: begin
                    busy <= 1'b1;
                    if (timer == INIT_CYCLES-1) begin
                        timer <= 16'd0;
                        init_calib_complete <= 1'b1;
                        busy <= 1'b0;
                        state <= S_IDLE;
                    end else begin
                        timer <= timer + 16'd1;
                    end
                end

                S_IDLE: begin
                    busy <= 1'b0;

                    // Automatic refresh takes the cycle only when refresh is due.
                    if (refresh_due) begin
                        busy <= 1'b1;
                        refresh_timer <= 16'd0;
                        refresh_event <= 1'b1;
                        refresh_pending <= 1'b1;
                        target_bank <= 2'd0;

                        if (|open_valid) begin
                            // Close the first currently-open bank, then refresh.
                            if (open_valid[0])       target_bank <= 2'd0;
                            else if (open_valid[1])  target_bank <= 2'd1;
                            else if (open_valid[2])  target_bank <= 2'd2;
                            else                     target_bank <= 2'd3;
                            state <= S_PRE;
                            timer <= 16'd0;
                        end else begin
                            state <= S_REFRESH;
                        end
                    end else if (cmd_valid) begin
                        busy         <= 1'b1;
                        target_bank <= addr_bank;
                        target_row  <= addr_row;
                        target_col  <= addr_col;
                        target_wdata<= cmd_wdata;
                        target_write<= cmd_write;
                        timer       <= 16'd0;

                        if (open_valid[addr_bank]) begin
                            if (open_row[addr_bank] == addr_row) begin
                                state <= cmd_write ? S_WRITE : S_READ;
                            end else begin
                                state <= S_PRE;
                            end
                        end else begin
                            state <= S_ACT;
                        end
                    end
                end

                S_PRE: begin
                    timer <= 16'd0;
                    state <= S_WAIT_RP;
                end


                S_WAIT_RP: begin
                    if (timer == T_RP-1) begin
                        timer <= 16'd0;
                        open_valid[target_bank] <= 1'b0;
                        if (refresh_pending) begin
                            state <= S_REFRESH;
                        end else begin
                            // Normal transaction is complete after the required
                            // precharge wait. Return response/ack and go idle.
                            state <= S_RESP;
                        end
                    end else begin
                        timer <= timer + 16'd1;
                    end
                end

// Top-level wrapper for the Digilent Arty A7-100T DDR3 MIG example design.
// CLK100MHZ is the board's 100 MHz oscillator on package pin E3.
// The Clocking Wizard generates:
//   clk_out1 = 166.666 MHz MIG system clock
//   clk_out2 = 200 MHz MIG IDELAY reference clock
// LED[0] lights after DDR3 calibration completes.
// LED[1] lights if the MIG traffic generator reports a compare/traffic error.

module arty_ddr_top (
    input  wire        CLK100MHZ,
    input  wire [2:0]  SW,
    output wire [1:0]  LED,

    inout  wire [15:0] ddr3_dq,
    inout  wire [1:0]  ddr3_dqs_n,
    inout  wire [1:0]  ddr3_dqs_p,

    output wire [13:0] ddr3_addr,
    output wire [2:0]  ddr3_ba,
    output wire        ddr3_ras_n,
    output wire        ddr3_cas_n,
    output wire        ddr3_we_n,
    output wire        ddr3_reset_n,
    output wire [0:0]  ddr3_ck_p,
    output wire [0:0]  ddr3_ck_n,
    output wire [0:0]  ddr3_cke,
    output wire [0:0]  ddr3_cs_n,
    output wire [1:0]  ddr3_dm,
    output wire [0:0]  ddr3_odt
);

    wire clk_sys_166;
    wire clk_ref_200;
    wire clock_locked;
    wire init_calib_complete;
    wire tg_compare_error;

    // The Clocking Wizard is configured for 100 MHz input,
    // 166.666 MHz output 1, 200 MHz output 2, and locked output enabled.
    arty_clock_wiz u_arty_clock_wiz (
        .clk_in1 (CLK100MHZ),
        .clk_out1(clk_sys_166),
        .clk_out2(clk_ref_200),
        .locked  (clock_locked)
    );

    // MIG's configured reset polarity is active-low. While the Clocking
    // Wizard is not locked, clock_locked=0 asserts sys_rst; once locked,
    // clock_locked=1 deasserts sys_rst.

        .sys_clk_i          (clk_sys_166),
        .clk_ref_i          (clk_ref_200),
        .sys_rst            (clock_locked),

        .ddr3_dq            (ddr3_dq),
        .ddr3_dqs_n          (ddr3_dqs_n),
        .ddr3_dqs_p          (ddr3_dqs_p),
        .ddr3_addr           (ddr3_addr),
        .ddr3_ba             (ddr3_ba),
        .ddr3_ras_n           (ddr3_ras_n),

        .ddr3_reset_n         (ddr3_reset_n),
        .ddr3_ck_p            (ddr3_ck_p),

        .ddr3_cs_n            (ddr3_cs_n),
